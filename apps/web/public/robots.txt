# Shared notes are private links, not public pages. They carry sensitive
# meeting content and are meant for one recipient, so they must never appear
# in a search index.
#
# This is defence in depth, not the control: the /s/** page sets a <meta robots
# noindex> in-app (SharedNote.tsx), vercel.json sends X-Robots-Tag: noindex on
# /s/**, and the shared-note API response sets X-Robots-Tag + Cache-Control:
# no-store. A crawler that ignores robots.txt still gets those.
User-agent: *
Disallow: /s/
Disallow: /api/

Allow: /
